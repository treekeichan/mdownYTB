cmake_minimum_required(VERSION 3.21)
project(mdownYTB VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
add_library(download_core src/downloadcore.cpp src/downloadcore.h src/engineprocess.cpp src/engineprocess.h)
target_link_libraries(download_core PUBLIC Qt6::Core)
target_include_directories(download_core PUBLIC src)
target_compile_definitions(download_core PUBLIC QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII NOMINMAX)
add_executable(mdownYTB WIN32 src/main.cpp src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(mdownYTB PRIVATE download_core Qt6::Widgets)
if(MSVC)
  target_compile_options(download_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(download_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(fake_engine tests/fake_engine.cpp)
  target_link_libraries(fake_engine PRIVATE Qt6::Core)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE download_core Qt6::Test)
  add_test(NAME core_tests COMMAND core_tests -o core-tests-results.txt,txt)
  add_executable(ui_tests tests/ui_tests.cpp src/mainwindow.cpp src/mainwindow.h)
  target_link_libraries(ui_tests PRIVATE download_core Qt6::Test Qt6::Widgets)
  add_dependencies(core_tests fake_engine)
  add_dependencies(ui_tests fake_engine)
  add_test(NAME ui_tests COMMAND ui_tests -platform offscreen -o ui-tests-results.txt,txt)
endif()
install(TARGETS mdownYTB RUNTIME DESTINATION .)
